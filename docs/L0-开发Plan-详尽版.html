开发 Plan · L0 数据底座 · 详尽版

L0 数据底座 — 详尽开发计划

把 L0 拆成 9 个 EPIC、约 35 个可独立交付的小功能，每个小功能都给出子任务清单、验收标准（DoD）、自测方法。可直接照此排期开工。核心信条不变：全量保存、可追溯、可重放。

范围：仅 L0（采集+存储+规范化+事件） EPIC：9 个 小功能：~35 个 预估：~23 人日（单人） 栈：Python · PG · 对象存储 · Redis/Arq

i文档说明与范围

本文档只覆盖 L0（数据底座）。不含 L1 分析、L2 判断、L3 产品——那些有各自的 Plan。L0 的唯一职责：把内容采进来、完整存下去、规范化、去重、发事件给上层。

结构：EPIC（大功能）→ Feature（小功能，可独立交付与验收）→ 子任务（实现单元）。每个 Feature 配 验收标准 与 自测方法。

优先级：MUST = MVP 必需；SHOULD = V1 完善。先做完所有 MUST 即得到可运行的数据底座。

▤数据库 DDL（L0，可执行）

-- 平台统一扩展（L0 表不用向量，L1 用）
CREATE EXTENSION IF NOT EXISTS vector;

-- 源注册表
CREATE TABLE sources(
  id BIGSERIAL PRIMARY KEY,
  name TEXT NOT NULL, feed_url TEXT, home_url TEXT,
  type TEXT NOT NULL CHECK (type IN ('article','podcast','video','tweet')),
  crawl_config JSONB NOT NULL DEFAULT '{}',   -- 选择器/周期/渲染开关
  etiquette JSONB NOT NULL DEFAULT '{}',      -- rate/concurrency/robots
  vertical_codes TEXT[] NOT NULL DEFAULT '{}',
  is_public BOOL NOT NULL DEFAULT true, owner_user_id BIGINT,
  enabled BOOL NOT NULL DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now());

CREATE TABLE source_health(
  source_id BIGINT PRIMARY KEY REFERENCES sources(id),
  last_ok_at TIMESTAMPTZ, last_attempt_at TIMESTAMPTZ,
  fail_count INT DEFAULT 0, empty_streak INT DEFAULT 0,
  status TEXT DEFAULT 'unknown', note TEXT);

CREATE TABLE crawl_jobs(
  id BIGSERIAL PRIMARY KEY, source_id BIGINT REFERENCES sources(id),
  kind TEXT NOT NULL, started_at TIMESTAMPTZ DEFAULT now(), finished_at TIMESTAMPTZ,
  status TEXT DEFAULT 'running', stats JSONB DEFAULT '{}', error TEXT);

-- ① 原始层（不可变，只追加）
CREATE TABLE raw_documents(
  id BIGSERIAL PRIMARY KEY, source_id BIGINT REFERENCES sources(id),
  crawl_job_id BIGINT REFERENCES crawl_jobs(id),
  url TEXT NOT NULL, fetch_method TEXT NOT NULL,
  http_status INT, http_headers JSONB, fetched_at TIMESTAMPTZ DEFAULT now(),
  raw_entry JSONB,            -- RSS/API 原始条目全字段
  raw_html_ref TEXT, raw_html_sha256 TEXT,  -- 对象存储指针 + 校验和
  media_refs JSONB DEFAULT '[]', extra JSONB DEFAULT '{}');
CREATE INDEX ON raw_documents(source_id, fetched_at);
CREATE INDEX ON raw_documents(raw_html_sha256);

-- ② 规范层（派生，可重建，上层主消费）
CREATE TABLE content_items(
  id BIGSERIAL PRIMARY KEY,
  canonical_url TEXT NOT NULL, url_hash TEXT NOT NULL UNIQUE,
  content_hash TEXT, simhash BIGINT,
  title TEXT, author TEXT, clean_text_ref TEXT, lang TEXT,
  source_id BIGINT REFERENCES sources(id), published_at TIMESTAMPTZ,
  fetched_at TIMESTAMPTZ DEFAULT now(), current_version INT DEFAULT 1,
  raw_document_id BIGINT REFERENCES raw_documents(id),
  status TEXT NOT NULL DEFAULT 'WAIT_FILTER',
  meta JSONB DEFAULT '{}', created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now());
CREATE INDEX ON content_items(status, fetched_at);
CREATE INDEX ON content_items(source_id, published_at);
CREATE INDEX ON content_items(content_hash);

-- 版本历史
CREATE TABLE content_versions(
  id BIGSERIAL PRIMARY KEY, content_id BIGINT REFERENCES content_items(id),
  version INT NOT NULL, raw_document_id BIGINT REFERENCES raw_documents(id),
  clean_text_ref TEXT, content_hash TEXT, diff_summary TEXT,
  detected_at TIMESTAMPTZ DEFAULT now(), UNIQUE(content_id, version));

-- ③ 事件发件箱（事务性 outbox，保证事件不丢）
CREATE TABLE outbox_events(
  id BIGSERIAL PRIMARY KEY, topic TEXT NOT NULL, payload JSONB NOT NULL,
  created_at TIMESTAMPTZ DEFAULT now(), published_at TIMESTAMPTZ);
CREATE INDEX ON outbox_events(published_at) WHERE published_at IS NULL;
设计落点：原始层(raw_documents) 与规范层(content_items) 通过 raw_document_id 强绑定可溯源；大文本只存对象存储指针 + sha256；outbox_events 用事务性发件箱保证"入库与发事件"原子一致、事件不丢。

F7.1content.ingested 事件（outbox）1d

✓ 验收

⚙ 自测

★L0 总验收（Definition of Done）

  • 采集：导入 OPML → 定时拉 RSS → 全文抓取，全链自动运行 ≥ 24h 无人工干预
  • 全量保存：每条内容的原始 entry / 原始 HTML / 抓取元数据完整留存，可取回校验
  • 规范化：content_items 状态正确（WAIT_FILTER），可溯源到 raw_document
  • 去重：转载/重复正确处理，原始层不丢变体
  • 版本：内容更新产生新版本，历史可查
  • 事件：content.ingested 不丢不重，L1 能稳定拉取
  • 可重放：能从原始层重建规范层，无需重抓外部
  • 质量：覆盖率达标，CI 全绿，/healthz 正常
  • 可观测：每次采集有 job 记录与指标（覆盖率/成功率/去重率）
